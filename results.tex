\begin{frame}
    \frametitle{Uranium Mass Supply}
    \begin{columns}
        \column[t]{5cm}
            \begin{itemize}
                \item All scenarios have the same uranium demands until 
                      advanced reactors are deployed
                \item Large peaks in Scenarios 2 and 5 correspond to the 
                      deployment of new \glspl{MMR}
                \item Less variation with time in the uranium supplied to reactors
                      for Scenarios 3 and 4
                \item Scenario 5 has the largest average mass of enriched urnaium 
                      sent to reactors
            \end{itemize}

\begin{frame}
    \frametitle{Feed Uranium Requirements}
    \begin{columns}
        \column[t]{5cm}
            \begin{itemize}
                \item Follows similar pattern to uranium mass, but at different 
                      magnitudes
                \item Scenario 2 requires the largest average mass of feed uranium
                \item The other scenarios require a feed uranium on a similar order 
                      of magnitude to \glspl{LWR}
                \item Between 2006-2020 the \gls{EIA} reports a maximum of 2,228 MT of U$_3$O$_8$ 
                      mined in a single year \cite{us_eia_2020_2021}
                
            \end{itemize}
        \column[t]{5cm}
        \vspace{-1cm}
        \begin{figure}
                \centering
                \includegraphics[height=0.4\textheight]{figures/feed_1-7.pdf}
                \caption{Feed uranium required to produce enriched uranium 
                for \glspl{LWR} (Scenario 1) and advanced reactors (Scenarios 2-7).}
                \label{fig:feed}
        \end{figure}
    \end{columns}
\end{frame}
